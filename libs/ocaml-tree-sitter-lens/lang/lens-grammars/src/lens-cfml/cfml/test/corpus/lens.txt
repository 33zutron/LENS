=====================================
Lens expression ellipsis in cfset
=====================================

<cfset x = foo(...)>

---

(program
  (cf_set_tag
    (assignment_expression
      (identifier)
      (call_expression
        (identifier)
        (arguments
          (lens_expression_ellipsis))))
    (cf_selfclose_void_tag_end)))
